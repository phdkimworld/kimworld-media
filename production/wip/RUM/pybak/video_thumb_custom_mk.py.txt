async def shot():
    from playwright.async_api import async_playwright
    async with async_playwright() as p:
        br = await p.chromium.launch(); pg = await br.new_page(viewport={'width': 1280, 'height': 720})
        for n in ('a', 'b', 'bg'):
            await pg.goto('file://' + os.path.join(D, n + '.html')); await pg.wait_for_timeout(300)
            await pg.screenshot(path=os.path.join(D, n + '.png'))
        await br.close()
asyncio.run(shot())
