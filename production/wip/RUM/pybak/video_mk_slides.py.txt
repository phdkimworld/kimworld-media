def st(f, title, big, label, tone='gold', sfx=None, note=None):
    s = {"from": f, "layout": "stat", "title": title, "stats": [{"at": f, "big": big, "label": label, "tone": tone}]}
    if note: s['stats'][0]['note'] = note
    if sfx: s['sfx'] = sfx
    return s
def q(f, text, sub=None, kicker=None, sfx=None):
    s = {"from": f, "layout": "quote", "text": text, "at": f}
    if sub: s.update(sub=sub, sub_at=f)
    if kicker: s['kicker'] = kicker
    if sfx: s['sfx'] = sfx
    return s
def cmp(f, title, lh, lb, ll, lt, rh, rb, rl, rt):
    return {"from": f, "layout": "compare", "title": title,
            "left": {"at": f, "head": lh, "big": lb, "lines": ll, "tone": lt},
            "right": {"at": f, "head": rh, "big": rb, "lines": rl, "tone": rt}}
def lst(f, title, items):
    return {"from": f, "layout": "list", "title": title, "items": [dict(it, at=it.get('at', f)) for it in items]}
